        // 从本地存储加载配置
        function loadConfig() {
            const savedConfig = localStorage.getItem('aiChatConfig');
            if (savedConfig) {
                try {
                    const parsedConfig = JSON.parse(savedConfig);
                    apiConfig = { ...apiConfig, ...parsedConfig };
                    
                    // 更新UI
                    apiProviderSelect.value = apiConfig.provider;
                    apiUrlInput.value = apiConfig.url;
                    apiKeyInput.value = apiConfig.apiKey;
                    temperatureInput.value = apiConfig.temperature;

                        populateModelSelect(apiConfig.availableModels, apiConfig.model);

                        enableChat();

                showPresetModels(apiConfig.provider);
            }

        // 填充模型选择下拉框
        function populateModelSelect(models, selectedModel = '') {
            // 清空现有选项
            modelSelect.innerHTML = '';
            
            if (models.length === 0) {
                const option = document.createElement('option');
                option.value = '';
                option.textContent = '未获取到模型列表';
                modelSelect.appendChild(option);
                return;
            }
            
            // 添加默认选项
            const defaultOption = document.createElement('option');
            defaultOption.value = '';
            defaultOption.textContent = '请选择模型';
            modelSelect.appendChild(defaultOption);
            
            // 添加模型选项
            models.forEach(model => {
                const option = document.createElement('option');
                option.value = model.id;
                option.textContent = model.name || model.id;
                
                // 如果是选中的模型，设置为选中
                if (selectedModel && (model.id === selectedModel || model.name === selectedModel)) {
                    option.selected = true;
                }
                
                modelSelect.appendChild(option);
            });
            
            // 如果有选中的模型但不在列表中，添加一个自定义选项
            if (selectedModel && !models.some(m => m.id === selectedModel || m.name === selectedModel)) {
                const customOption = document.createElement('option');
                customOption.value = selectedModel;
                customOption.textContent = `${selectedModel} (自定义)`;
                customOption.selected = true;
                modelSelect.appendChild(customOption);
            }
        }
        
        // 获取模型列表
        async function fetchModels() {
            const apiKey = apiKeyInput.value.trim();
            const provider = apiProviderSelect.value;
            
            if (!apiKey) {
                showModelMessage('请输入API密钥', 'error');
                return;
            }
            
            // 显示加载状态
            modelLoading.style.display = 'flex';
            modelMessage.style.display = 'none';
            fetchModelsBtn.disabled = true;
            
            try {
                let models = [];
                
                // 根据提供商使用不同的方法获取模型
                switch(provider) {
                    case 'openai':
                        models = await fetchOpenAIModels(apiKey);
                        break;
                    case 'anthropic':
                        models = await fetchAnthropicModels(apiKey);
                        break;
                    case 'google':
                        models = await fetchGoogleModels(apiKey);
                        break;
                    case 'custom':
                        // 对于自定义API，尝试推断模型列表端点
                        const customModels = await fetchCustomModels(apiKey, apiUrlInput.value.trim());
                        if (customModels && customModels.length > 0) {
                            models = customModels;
                        } else {
                            throw new Error('无法自动获取模型列表，请使用预定义模型或手动输入');
                        }
                        break;
                }
                
                // 保存到配置
                apiConfig.availableModels = models;
                apiConfig.provider = provider;
                
                // 更新UI
                populateModelSelect(models);
                
                // 显示成功消息
                showModelMessage(`成功获取 ${models.length} 个可用模型`, 'success');
                
                // 隐藏预定义模型
                presetModels.style.display = 'none';
                
            } catch (error) {
                console.error('获取模型列表失败:', error);
                showModelMessage(`获取模型列表失败: ${error.message}`, 'error');
                
                // 显示预定义模型作为备选
                showPresetModels(provider);
                
                // 如果之前有保存的模型列表，显示它们
                if (apiConfig.availableModels && apiConfig.availableModels.length > 0) {
                    populateModelSelect(apiConfig.availableModels);
                }
            } finally {
                modelLoading.style.display = 'none';
                fetchModelsBtn.disabled = false;
            }
        }
        
        // 获取OpenAI模型列表
        async function fetchOpenAIModels(apiKey) {
            const modelsUrl = 'https://api.openai.com/v1/models';
            
            const response = await fetch(modelsUrl, {
                method: 'GET',
                headers: {
                    'Authorization': `Bearer ${apiKey}`
                }
            });
            
            if (!response.ok) {
                throw new Error(`OpenAI API请求失败: ${response.status}`);
            }
            
            const data = await response.json();
            
            // 过滤出聊天模型并排序
            return data.data
                .filter(model => 
                    model.id.includes('gpt') && 
                    !model.id.includes('instruct') && 
                    !model.id.includes('search') && 
                    !model.id.includes('edit')
                )
                .map(model => ({
                    id: model.id,
                    name: model.id
                }))
                .sort((a, b) => {
                    // 将gpt-4系列排在前面
                    if (a.id.includes('gpt-4') && !b.id.includes('gpt-4')) return -1;
                    if (!a.id.includes('gpt-4') && b.id.includes('gpt-4')) return 1;
                    return a.id.localeCompare(b.id);
                });
        }
        
        // 获取Anthropic模型列表
        async function fetchAnthropicModels(apiKey) {
            const modelsUrl = 'https://api.anthropic.com/v1/models';
            
            const response = await fetch(modelsUrl, {
                method: 'GET',
                headers: {
                    'Authorization': `Bearer ${apiKey}`,
                    'anthropic-version': '2023-06-01'
                }
            });
            
            if (!response.ok) {
                // Anthropic可能没有公开的模型列表端点，返回预定义列表
                if (response.status === 404) {
                    return presetModelList.anthropic.filter(m => m.id);
                }
                throw new Error(`Anthropic API请求失败: ${response.status}`);
            }
            
            const data = await response.json();
            
            // 提取模型数据
            return data.data.map(model => ({
                id: model.id,
                name: model.name || model.id
            })).sort((a, b) => a.name.localeCompare(b.name));
        }
        
        // 获取Google模型列表
        async function fetchGoogleModels(apiKey) {
            // Google的模型列表端点
            const modelsUrl = 'https://generativelanguage.googleapis.com/v1beta/models?key=' + apiKey;
            
            const response = await fetch(modelsUrl, {
                method: 'GET'
            });
            
            if (!response.ok) {
                throw new Error(`Google API请求失败: ${response.status}`);
            }
            
            const data = await response.json();
            
            // 提取模型数据
            return data.models
                .filter(model => model.name && model.name.includes('gemini'))
                .map(model => ({
                    id: model.name.replace('models/', ''),
                    name: model.displayName || model.name.replace('models/', '')
                }))
                .sort((a, b) => a.name.localeCompare(b.name));
        }
        
        // 获取自定义API模型列表
        async function fetchCustomModels(apiKey, apiUrl) {
            if (!apiUrl) {
                throw new Error('请输入API端点URL');
            }
            
            // 尝试从标准端点推断模型列表端点
            let modelsUrl = apiUrl;
            
            if (modelsUrl.includes('/chat/completions')) {
                modelsUrl = modelsUrl.replace('/chat/completions', '/models');
            } else if (modelsUrl.includes('/v1/messages')) {
                modelsUrl = modelsUrl.replace('/v1/messages', '/v1/models');
            } else if (modelsUrl.includes('/generateContent')) {
                modelsUrl = modelsUrl.replace('/generateContent', '');
                if (!modelsUrl.endsWith('/models')) {
                    modelsUrl = modelsUrl.replace(/\/$/, '') + '/models';
                }
            } else {
                // 如果无法推断，尝试直接请求提供的端点
                // 先测试是否是对话端点
                try {
                    const testResponse = await fetch(apiUrl, {
                        method: 'POST',
                        headers: {
                            'Authorization': `Bearer ${apiKey}`,
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify({
                            model: 'test',
                            messages: [{ role: 'user', content: 'test' }],
                            max_tokens: 1
                        })
                    });
                    
                    if (testResponse.status === 404 || testResponse.status === 400) {
                        // 可能是模型列表端点
                        modelsUrl = apiUrl;
                    } else {
                        throw new Error('无法确定模型列表端点，请手动输入模型名称');
                    }
                } catch (e) {
                    throw new Error('无法连接到自定义API，请检查URL和密钥');
                }
            }
            
            // 尝试获取模型列表
            try {
                const response = await fetch(modelsUrl, {
                    method: 'GET',
                    headers: {
                        'Authorization': `Bearer ${apiKey}`
                    }
                });
                
                if (!response.ok) {
                    throw new Error(`自定义API请求失败: ${response.status}`);
                }
                
                const data = await response.json();
                
                // 尝试解析不同格式的响应
                let models = [];
                if (data.data && Array.isArray(data.data)) {
                    models = data.data.map(model => ({
                        id: model.id || model.name,
                        name: model.name || model.id
                    }));
                } else if (data.models && Array.isArray(data.models)) {
                    models = data.models.map(model => ({
                        id: model.id || model.name,
                        name: model.name || model.id
                    }));
                } else if (Array.isArray(data)) {
                    models = data.map(model => ({
                        id: model.id || model.name || model,
                        name: model.name || model.id || model
                    }));
                } else {
                    throw new Error('无法解析模型列表响应格式');
                }
                
                return models;
                
            } catch (error) {
                // 如果获取失败，使用预定义的自定义模型选项
                throw new Error('无法自动获取模型列表，请使用预定义模型或手动输入');
            }
        }

        // 发送消息到AI
        async function sendMessageToAI(message) {
            if (!apiConfig.connected) {
                addMessage('系统', '请先配置API设置并连接', true);
                return;
            }
            
            if (!apiConfig.model) {
                addMessage('系统', '请先选择AI模型', true);
                return;
            }
            
            showTypingIndicator();
            
            try {
                // 根据不同的API提供商构建请求
                let requestBody = {};
                let headers = {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${apiConfig.apiKey}`
                };
                
                if (apiConfig.provider === 'openai') {
                    requestBody = {
                        model: apiConfig.model,
                        messages: [{ role: 'user', content: message }],
                        temperature: parseFloat(apiConfig.temperature),
                        max_tokens: 1000
                    };
                } else if (apiConfig.provider === 'anthropic') {
                    headers['anthropic-version'] = '2023-06-01';
                    headers['x-api-key'] = apiConfig.apiKey;
                    delete headers['Authorization'];
                    
                    requestBody = {
                        model: apiConfig.model,
                        messages: [{ role: 'user', content: message }],
                        max_tokens: 1000,
                        temperature: parseFloat(apiConfig.temperature)
                    };
                } else if (apiConfig.provider === 'google') {
                    // Google Gemini API 格式
                    // 注意：Google API使用API密钥作为查询参数，而不是Authorization头
                    const urlWithKey = apiConfig.url.includes('?') 
                        ? `${apiConfig.url}&key=${apiConfig.apiKey}`
                        : `${apiConfig.url}?key=${apiConfig.apiKey}`;
                    
                    requestBody = {
                        contents: [{
                            parts: [{ text: message }]
                        }],
                        generationConfig: {
                            temperature: parseFloat(apiConfig.temperature),
                            maxOutputTokens: 1000
                        }
                    };
                    
                    // 发送请求到Google API
                    const response = await fetch(urlWithKey, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify(requestBody)
                    });
                    
                    if (!response.ok) {
                        throw new Error(`Google API请求失败: ${response.status} ${response.statusText}`);
                    }
                    
                    const data = await response.json();
                    const aiResponse = data.candidates[0].content.parts[0].text;
                    
                    hideTypingIndicator();
                    addMessage('AI助手', aiResponse, true);
                    return;
                    
                } else {
                    // 自定义API - 假设使用OpenAI格式
                    requestBody = {
                        model: apiConfig.model,
                        messages: [{ role: 'user', content: message }],
                        temperature: parseFloat(apiConfig.temperature)
                    };
                }
                
                const response = await fetch(apiConfig.url, {
                    method: 'POST',
                    headers: headers,
                    body: JSON.stringify(requestBody)
                });
                
                if (!response.ok) {
                    throw new Error(`API请求失败: ${response.status} ${response.statusText}`);
                }
                
                const data = await response.json();
                
                // 解析不同API提供商的响应
                let aiResponse = '';
                if (apiConfig.provider === 'openai') {
                    aiResponse = data.choices[0].message.content;
                } else if (apiConfig.provider === 'anthropic') {
                    aiResponse = data.content[0].text;
                } else {
                    // 自定义API - 假设使用OpenAI格式
                    aiResponse = data.choices?.[0]?.message?.content || 
                                 data.response || 
                                 JSON.stringify(data);
                }
                
                hideTypingIndicator();
                addMessage('AI助手', aiResponse, true);
                
            } catch (error) {
                hideTypingIndicator();
                console.error('AI请求错误:', error);
                addMessage('系统', `请求AI时出错: ${error.message}`, true);
            }
        }

            addMessage('您', message, false);
